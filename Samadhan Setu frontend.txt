    let score = 0;
    const urgentWords = ['danger', 'accident', 'fire', 'spark', 'death', 'injury', 'urgent', 'blocked'];

    if (category === 'Health & Safety' || category === 'Electricity') score += 2;
    if (upvotes > 20) score += 3;
    
    if (score >= 5) return 'High';
    if (score >= 2) return 'Medium';
    return 'Low';

                            <Badge status={g.status} />
                            {g.priority === 'High' && (
                                <span className="flex items-center gap-1 text-xs font-bold text-red-600">
                                    <AlertTriangle size={12}/> High Urgency
                                </span>
                            )}
                        </div>
                    </div>
                </div>
            ))}
        </div>
    </div>
  );

  const DeptDashboard = () => {
    // Filter grievances for this officer's department
    // In this mock, the officer is 'Roads' dept, but let's just show relevant ones + general for demo
    const deptGrievances = grievances.filter(g => g.category === 'Roads & Transport' || g.status !== 'Resolved').sort((a,b) => {
        const priorityWeight = { 'High': 3, 'Medium': 2, 'Low': 1 };
        return priorityWeight[b.priority] - priorityWeight[a.priority];
    });

    return (
        <div className="max-w-6xl mx-auto my-8 px-4">
             <div className="flex justify-between items-end mb-6">
                <div>
                    <h2 className="text-2xl font-bold text-gray-800">Department Dashboard</h2>
                    <p className="text-sm text-gray-500">Welcome, Officer. You have <span className="font-bold text-orange-600">{deptGrievances.filter(g => g.status !== 'Resolved').length} pending</span> tasks.</p>
                </div>
                <div className="bg-white p-2 rounded border flex gap-4 text-sm">
                    <div className="flex items-center gap-2"><div className="w-3 h-3 rounded-full bg-red-100 border border-red-500"></div> High</div>
                    <div className="flex items-center gap-2"><div className="w-3 h-3 rounded-full bg-orange-100 border border-orange-500"></div> Medium</div>
                    <div className="flex items-center gap-2"><div className="w-3 h-3 rounded-full bg-blue-100 border border-blue-500"></div> Low</div>
                </div>
             </div>

             <div className="bg-white rounded shadow overflow-hidden border border-gray-200">
                <table className="w-full text-left border-collapse">
                    <thead className="bg-gray-100 text-gray-600 text-xs uppercase font-semibold">
                        <tr>
                            <th className="p-4 border-b">ID</th>
                            <th className="p-4 border-b">Priority (AI)</th>
                            <th className="p-4 border-b">Issue / Location</th>
                            <th className="p-4 border-b">Votes</th>
                            <th className="p-4 border-b">Status</th>
                            <th className="p-4 border-b">Action</th>
                        </tr>
                    </thead>
                    <tbody className="text-sm text-gray-700">
                        {deptGrievances.map(g => (
                            <tr key={g.id} className="hover:bg-gray-50 border-b last:border-0">
                                <td className="p-4 font-mono">#{g.id}</td>
                                <td className="p-4">
                                    <span className={`flex items-center gap-2 font-bold ${g.priority === 'High' ? 'text-red-600' : 'text-gray-600'}`}>
                                        {g.priority === 'High' && <AlertTriangle size={14}/>} {g.priority}
                                    </span>
                                </td>
                                <td className="p-4">
                                    <div className="font-medium text-gray-900 mb-1">{g.category}</div>
                                    <div className="truncate max-w-xs" title={g.description}>{g.description}</div>
                                    <div className="text-xs text-gray-400 mt-1">{g.location}</div>
                                </td>
                                <td className="p-4 font-semibold">{g.upvotes}</td>
                                <td className="p-4"><Badge status={g.status} /></td>
                                <td className="p-4">
                                    {g.status !== 'Resolved' ? (
                                        <button 
                                            onClick={() => updateStatus(g.id, 'Resolved')}
                                            className="bg-green-600 hover:bg-green-700 text-white px-3 py-1 rounded text-xs shadow-sm transition"
                                        >
                                            Mark Resolved
                                        </button>
                                    ) : (
                                        <span className="text-green-600 flex items-center gap-1 text-xs"><CheckCircle size={14}/> Done</span>
                                    )}
                                </td>
                            </tr>
                        ))}
                    </tbody>
                </table>
             </div>
        </div>
    );
  };

  const AdminDashboard = () => {
    // Simple stats
    const total = grievances.length;
    const pending = grievances.filter(g => g.status === 'Pending').length;
    const resolved = grievances.filter(g => g.status === 'Resolved').length;

    return (
        <div className="max-w-6xl mx-auto my-8 px-4">
            <h2 className="text-2xl font-bold text-gray-800 mb-6">Administration Overview</h2>
            
            <div className="grid grid-cols-1 md:grid-cols-4 gap-6 mb-8">
                <div className="bg-white p-6 rounded shadow-sm border-l-4 border-blue-500">
                    <div className="text-gray-500 text-sm font-semibold uppercase">Total Grievances</div>
                    <div className="text-3xl font-bold text-gray-800 mt-2">{total}</div>
                </div>
                <div className="bg-white p-6 rounded shadow-sm border-l-4 border-red-500">
                    <div className="text-gray-500 text-sm font-semibold uppercase">Pending Action</div>
                    <div className="text-3xl font-bold text-red-600 mt-2">{pending}</div>
                </div>
                <div className="bg-white p-6 rounded shadow-sm border-l-4 border-green-500">
                    <div className="text-gray-500 text-sm font-semibold uppercase">Resolved</div>
                    <div className="text-3xl font-bold text-green-600 mt-2">{resolved}</div>
                </div>
                <div className="bg-white p-6 rounded shadow-sm border-l-4 border-orange-500">
                    <div className="text-gray-500 text-sm font-semibold uppercase">Citizen Engagement</div>
                    <div className="text-3xl font-bold text-orange-600 mt-2">1.2k <span className="text-xs text-gray-400 font-normal">Active Users</span></div>
                </div>
            </div>

            <div className="grid md:grid-cols-2 gap-8">
                <div className="bg-white p-6 rounded shadow border border-gray-200">
                    <h3 className="font-bold text-gray-800 mb-4 flex items-center gap-2"><BarChart3 size={20} /> Category Breakdown</h3>
                    <div className="space-y-4">
                        {DEPARTMENTS.map((d, i) => {
                            const count = grievances.filter(g => g.category === d.name).length;
                            const pct = total ? (count / total) * 100 : 0;
                            return (
                                <div key={d.id}>
                                    <div className="flex justify-between text-sm mb-1">
                                        <span>{d.name}</span>
                                        <span className="font-bold">{count}</span>
                                    </div>
                                    <div className="w-full bg-gray-100 rounded-full h-2">
                                        <div className="bg-blue-600 h-2 rounded-full" style={{ width: `${pct}%` }}></div>
                                    </div>
                                </div>
                            );
                        })}
                    </div>
                </div>

                <div className="bg-white p-6 rounded shadow border border-gray-200">
                    <h3 className="font-bold text-gray-800 mb-4 flex items-center gap-2"><AlertTriangle size={20} /> High Priority Zones</h3>
                    <div className="space-y-3">
                         {/* Mock Data for zones */}
                        {['Ward 12 - Market Area', 'Sector 4 - Housing Board', 'MG Road - Commercial'].map((zone, i) => (
                            <div key={i} className="flex items-center justify-between p-3 bg-red-50 rounded border border-red-100">
                                <span className="text-sm font-medium text-gray-800">{zone}</span>
                                <span className="text-xs font-bold text-red-600 bg-white px-2 py-1 rounded border border-red-200">Critical</span>
                            </div>
                        ))}
                    </div>
                </div>
            </div>
        </div>
    );
  };

  return (
    <div className="font-sans text-gray-800 min-h-screen flex flex-col bg-gray-50">
      <Navbar user={user} onLogout={() => { setUser(null); setView('home'); }} setView={setView} />
      
      <main className="flex-grow">
        {view === 'home' && <HomeView />}
        {view === 'login' && <LoginView />}
        {view === 'lodge' && <LodgeGrievance />}
        {view === 'track' && <TrackGrievance />}
        {view === 'community' && <CommunityFeed />}
        {view === 'dept_dashboard' && <DeptDashboard />}
        {view === 'admin_dashboard' && <AdminDashboard />}
      </main>

      {/* Footer */}
      <footer className="bg-gray-800 text-white pt-10 pb-4 text-sm mt-auto">
        <div className="max-w-6xl mx-auto px-4 grid grid-cols-1 md:grid-cols-4 gap-8 mb-8">
            <div>
                <h4 className="font-bold text-lg mb-4 text-gray-200">Samadhan Setu</h4>
                <p className="text-gray-400 text-xs leading-relaxed">
                    An initiative to bridge the gap between citizens and administration using advanced AI technologies for a transparent and efficient governance.
                </p>
            </div>
            <div>
                <h4 className="font-bold mb-4 text-gray-300">Quick Links</h4>
                <ul className="space-y-2 text-gray-400 text-xs">
                    <li className="hover:text-white cursor-pointer">About Us</li>
                    <li className="hover:text-white cursor-pointer">Contact Administration</li>
                    <li className="hover:text-white cursor-pointer">Privacy Policy</li>
                    <li className="hover:text-white cursor-pointer">Terms of Service</li>
                </ul>
            </div>
            <div>
                <h4 className="font-bold mb-4 text-gray-300">Contact Us</h4>
                <ul className="space-y-2 text-gray-400 text-xs">
                    <li className="flex items-center gap-2"><Phone size={14}/> 1800-11-0000 (Toll Free)</li>
                    <li className="flex items-center gap-2"><MapPin size={14}/> Ministry of Urban Affairs, New Delhi</li>
                </ul>
            </div>
            <div>
                <h4 className="font-bold mb-4 text-gray-300">Accessibility</h4>
                <div className="flex gap-2 text-xs text-gray-400">
                    <span className="border border-gray-600 px-2 py-1 rounded">Screen Reader</span>
                    <span className="border border-gray-600 px-2 py-1 rounded">High Contrast</span>
                </div>
            </div>
        </div>
        <div className="text-center text-xs text-gray-500 border-t border-gray-700 pt-4">
            © 2024 Samadhan Setu | Government of India. All rights reserved.
        </div>
      </footer>
    </div>
  );
}
